{{- /* Category icon for a post, keyed off its first tag. On single posts (link=true) it's a link to the tag page — giving the icon a purpose and the hover animation a natural target; in lists it stays decorative (the card itself is already the link). */ -}} {{- $page := .page }} {{- $link := .link }} {{- with $page.Params.tags }} {{- $tag := index . 0 }} {{- $key := $tag | urlize }} {{- with (resources.Get (printf "icons/%s.svg" $key)) }} {{- if $link }} {{ .Content | safeHTML }} {{- else }} {{ .Content | safeHTML }} {{- end }} {{- end }} {{- end }}
